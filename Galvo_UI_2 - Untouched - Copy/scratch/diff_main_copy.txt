+            old_end = getattr(self, '_last_total_passes', passes)
+            try:
+                end_l = int(self.input_3d_end_layer.text() or passes)
+            except ValueError:
+                end_l = passes
+            if end_l == old_end or end_l > passes or end_l <= 0:
+                end_l = passes
+                self.input_3d_end_layer.blockSignals(True)
+                self.input_3d_end_layer.setText(str(end_l))
+                self.input_3d_end_layer.blockSignals(False)
+
+        self._last_total_passes = passes
+
+        start_l = max(1, min(passes, start_l))
+        end_l = max(start_l, min(passes, end_l))
+        passes_to_print = end_l - start_l + 1
+
+        if hasattr(self, 'label_3d_calculated_passes'):
+            if start_l == 1 and end_l == passes:
+                self.label_3d_calculated_passes.setText(f"{passes} Passes ({z_step:.3f} mm/pass) | Depth: {depth:.2f} mm")
+            else:
+                self.label_3d_calculated_passes.setText(f"{passes_to_print} Passes (Layers {start_l} to {end_l} of {passes}) | {z_step:.3f} mm/pass")
+
+        # Sync Loop Count widget with the actual number of layers to print
+        if hasattr(self, 'programgalvo_widgets') and hasattr(self.programgalvo_widgets, 'pgmloopcount'):
+            self.wh.invokeMethod(self.programgalvo_widgets.pgmloopcount, "set", str(passes_to_print))

+    def on_3d_preview_slider_changed(self, value):
+        if not hasattr(self, 'stl_engraving_mgr') or self.stl_engraving_mgr.cached_mesh is None:

+            return

+        # Keep Preview page slider in sync
+        if hasattr(self, 'slider_preview_3d_layer'):
+            self.slider_preview_3d_layer.blockSignals(True)
+            self.slider_preview_3d_layer.setValue(value)
+            self.slider_preview_3d_layer.blockSignals(False)

+        range_str = ""

+        # Update 3D visual cutting plane immediately (<0.01ms with GPU display list)

+        layer_idx = max(0, min(total_layers - 1, value - 1))
+        self._pending_preview_layer_idx = layer_idx

+        # If 2D preview is hidden (e.g. 3D Model mode), skip 2D toolpath generation completely
+        if hasattr(self, 'galvo_preview') and not self.galvo_preview.isVisible():
+            return
+
+        # Check in-memory cache for this layer
+        if not hasattr(self, '_layer_queue_cache'):
+            self._layer_queue_cache = {}
+
+        if layer_idx in self._layer_queue_cache:
+            # Cached: instant 0ms draw
+            cached_queue = self._layer_queue_cache[layer_idx]
+            self.planner.clear()
+            self.planner.preview_queue = list(cached_queue)
+            self.planner.send_queue = list(cached_queue)
+            if hasattr(self, "galvo_preview") and self.galvo_preview.isVisible():
+                self.galvo_preview.draw_queue(self.planner.preview_queue)
+            return
+
+        # Not cached: debounce heavy Shapely hatching calculation to prevent UI freeze during smooth scrubbing
+        if hasattr(self, '_slider_2d_debounce_timer'):
+            self._slider_2d_debounce_timer.start(50)
+        else:
+            self._render_active_preview_layer_hatch()

+    def _render_active_preview_layer_hatch(self):
+        """Debounced generator and renderer of 2D toolpath hatching for the active layer."""
+        if not hasattr(self, 'stl_engraving_mgr') or self.stl_engraving_mgr.cached_mesh is None:
+            return
+        if not getattr(self.stl_engraving_mgr, 'cached_layer_polys', None):
+            return

+            return

+        total_layers = len(self.stl_engraving_mgr.cached_layer_polys) or 1
+        curr_val = self.slider_preview_3d_layer.value() if hasattr(self, 'slider_preview_3d_layer') else 1
+        layer_idx = getattr(self, '_pending_preview_layer_idx', max(0, min(total_layers - 1, curr_val - 1)))
+
+        if not hasattr(self, '_layer_queue_cache'):
+            self._layer_queue_cache = {}
+
+        if layer_idx in self._layer_queue_cache:
+            layer_queue = self._layer_queue_cache[layer_idx]
+        else:
+            try:
+                hatch_pitch = float(self.input_3d_hatch_pitch.text() or 0.035)
+                cross_rot = self.cb_3d_cross_layer_rotate.isChecked() if hasattr(self, 'cb_3d_cross_layer_rotate') else True
+                start_x = float(self.wh.invokeMethod(self.programgalvo_widgets.pgmstartposx, "get") or 0.0)
+                start_y = float(self.wh.invokeMethod(self.programgalvo_widgets.pgmstartposy, "get") or 0.0)
+                galvo_units_per_mm = 65535 / self.galvo_config.field_size
+                raw_jump = float(self.wh.invokeMethod(self.programgalvo_widgets.pgmjumpspeed, "get") or 2000.0)

+                layer_queue = self.stl_engraving_mgr.generate_layer_queue(
+                    layer_idx=layer_idx,
+                    galvo_config=self.galvo_config,
+                    hatch_pitch_mm=hatch_pitch,
+                    cross_layer_rotate=cross_rot,
+                    mark_speed=mark_spd,
+                    jump_speed=jump_spd,
+                    start_x_mm=start_x,
+                    start_y_mm=start_y,
+                    mark_contour=mark_cnt
+                )
+                if len(self._layer_queue_cache) > 300:
+                    self._layer_queue_cache.clear()
+                self._layer_queue_cache[layer_idx] = layer_queue
+            except Exception as e:
+                print(f"[3D Preview Error] {e}")
+                return
+
+        self.planner.clear()
+        self.planner.preview_queue = list(layer_queue)
+        self.planner.send_queue = list(layer_queue)
+        if hasattr(self, "galvo_preview") and self.galvo_preview.isVisible():
+            self.galvo_preview.draw_queue(self.planner.preview_queue)

+    def load_3d_stl_info(self, filepath):
+        """
+        Loads the STL model metadata and prepares dimensions, scale, and pass calculation
+        WITHOUT slicing the model. Slicing only occurs when SAVE is clicked.
+        """
+        try:
+            ok = self.stl_engraving_mgr.load_stl(filepath)
+            if not ok:
+                self.showCustomPopup("Invalid STL File", "Could not load or parse the selected STL file.", buttons=QMessageBox.Ok)
+                return False
+
+            # If user had set a scale previously (e.g., loaded from json), apply it
+            if hasattr(self, 'input_3d_scale') and self.input_3d_scale.text():
+                try:
+                    s_pct = float(self.input_3d_scale.text())
+                    if s_pct > 0 and abs(s_pct - 100.0) > 0.01:
+                        self.stl_engraving_mgr.apply_scale(s_pct / 100.0)
+                except Exception:
+                    pass
+
+            dim_x, dim_y, dim_z = self.stl_engraving_mgr.get_mesh_dimensions()
+            if hasattr(self, 'label_3d_stl_dims'):
+                self.label_3d_stl_dims.setText(f"X: {dim_x:.1f} mm | Y: {dim_y:.1f} mm | Z: {dim_z:.2f} mm")
+            if hasattr(self, 'input_3d_scale') and not self.input_3d_scale.text():
+                self.input_3d_scale.setText(f"{self.stl_engraving_mgr.scale_factor * 100.0:.1f}")
+
+            try:
+                z_step = float(self.input_3d_z_step.text() or 0.015)
+            except Exception:
+                z_step = 0.015
+
+            passes = self.stl_engraving_mgr.calculate_passes(dim_z, z_step)
+            if hasattr(self, 'input_3d_start_layer'):
+                if not self.input_3d_start_layer.text() or int(self.input_3d_start_layer.text() or 1) < 1:
+                    self.input_3d_start_layer.setText("1")
+            if hasattr(self, 'input_3d_end_layer'):
+                self.input_3d_end_layer.setText(str(passes))
+            self.update_3d_calculations()
+
+            self._layer_queue_cache = {}
+            if hasattr(self, 'stl_3d_preview'):
+                self.stl_3d_preview.load_mesh(self.stl_engraving_mgr.cached_mesh)
+
+            return True
+        except Exception as e:
+            print(f"[load_3d_stl_info] Error: {e}")
+            return False

+    def process_3d_stl_file(self, filepath, lens_size, galvo_units_per_mm, jump_speed, mark_speed, start_x, start_y):
+        try:
+            ok = self.stl_engraving_mgr.load_stl(filepath)
+            if not ok:
+                self.showCustomPopup("Invalid STL File", "Could not load or parse the selected STL file.", buttons=QMessageBox.Ok)
+                return False
+
+            # Ensure any scale from the UI is applied to the mesh before slicing
+            if hasattr(self, 'input_3d_scale'):
+                try:
+                    target_scale_pct = float(self.input_3d_scale.text() or 100.0)
+                    curr_scale_pct = self.stl_engraving_mgr.scale_factor * 100.0
+                    if abs(target_scale_pct - curr_scale_pct) > 0.01 and target_scale_pct > 0:
+                        mult = target_scale_pct / max(0.001, curr_scale_pct)
+                        self.stl_engraving_mgr.apply_scale(mult)
+                except Exception:
+                    pass
+
+            dim_x, dim_y, dim_z = self.stl_engraving_mgr.get_mesh_dimensions()
+            try:
+                bg_margin = float(self.input_3d_bg_margin.text() or 2.0)
+            except Exception:
+                bg_margin = 2.0
+            try:
+                z_step = float(self.input_3d_z_step.text() or 0.015)
+            except Exception:
+                z_step = 0.015
+            carve_mode = self.combo_3d_carve_mode.currentText() if hasattr(self, 'combo_3d_carve_mode') else STL3DEngravingManager.MODE_RELIEF
+
+            total_w = dim_x + 2 * bg_margin
+            total_h = dim_y + 2 * bg_margin
+
+            if total_w > lens_size or total_h > lens_size:
+                from PySide6.QtWidgets import QMessageBox
+                reply = QMessageBox.question(
+                    self,
+                    "Auto-Fit 3D Relief to Lens",
+                    f"Total work area including margin ({total_w:.1f} x {total_h:.1f} mm) exceeds the lens area ({lens_size:.1f} x {lens_size:.1f} mm).\n\n"
+                    f"Would you like to automatically scale it down to fit the working area?",
+                    QMessageBox.Yes | QMessageBox.No,
+                    QMessageBox.Yes
+                )
+                if reply == QMessageBox.Yes:
+                    self.stl_engraving_mgr.auto_fit_to_lens(lens_size, bg_margin)
+                    dim_x, dim_y, dim_z = self.stl_engraving_mgr.get_mesh_dimensions()
+                    total_w = dim_x + 2 * bg_margin
+                    total_h = dim_y + 2 * bg_margin
+                    if hasattr(self, 'input_3d_scale'):
+                        self.input_3d_scale.blockSignals(True)
+                        self.input_3d_scale.setText(f"{self.stl_engraving_mgr.scale_factor * 100.0:.1f}")
+                        self.input_3d_scale.blockSignals(False)
+                else:
+                    return False
+
+            max_shift_x = max(0.0, (lens_size - total_w) / 2.0)
+            max_shift_y = max(0.0, (lens_size - total_h) / 2.0)
+            if abs(start_x) > max_shift_x or abs(start_y) > max_shift_y:
+                start_x = max(-max_shift_x, min(max_shift_x, start_x))
+                start_y = max(-max_shift_y, min(max_shift_y, start_y))
+                if hasattr(self, 'programgalvo_widgets'):
+                    if hasattr(self.programgalvo_widgets, 'pgmstartposx'):
+                        self.wh.invokeMethod(self.programgalvo_widgets.pgmstartposx, "set", f"{start_x:.1f}")
+                    if hasattr(self.programgalvo_widgets, 'pgmstartposy'):
+                        self.wh.invokeMethod(self.programgalvo_widgets.pgmstartposy, "set", f"{start_y:.1f}")
+
+            # Slice the model top-down (cached and super-fast)
+            layers = self.stl_engraving_mgr.slice_model(
+                z_step_mm=z_step,
+                target_depth_mm=dim_z,
+                bg_margin_mm=bg_margin,
+                carve_mode=carve_mode
+            )
+            total_passes = len(layers)
+
+            # START & END LAYER RANGE
+            start_l = 1
+            if hasattr(self, 'input_3d_start_layer') and self.input_3d_start_layer.text():
+                try:
+                    start_l = int(self.input_3d_start_layer.text())
+                except ValueError:
+                    start_l = 1
+            end_l = total_passes
+            if hasattr(self, 'input_3d_end_layer') and self.input_3d_end_layer.text():
+                try:
+                    end_l = int(self.input_3d_end_layer.text())
+                except ValueError:
+                    end_l = total_passes
+
+            start_l = max(1, min(total_passes, start_l))
+            end_l = max(start_l, min(total_passes, end_l))
+            passes_to_print = end_l - start_l + 1
+
+            # Update UI labels
+            if hasattr(self, 'label_3d_stl_dims'):
+                self.label_3d_stl_dims.setText(f"X: {dim_x:.1f} mm | Y: {dim_y:.1f} mm | Z: {dim_z:.2f} mm")

+    def update_split_preset_options(self, width: float = 0.0, height: float = 0.0, lens_size: float = None):
